      Programa(Context c, float fuerza) { super(false, 1); glProgram = new GlProgram(c,
        "shaders/suavizado.vert", "shaders/suavizado.frag"); ... }
      // el significado exacto de (boolean, int) del constructor: fuentes de Media3 1.9.0 -P9-
      @Override public Size configure(int w, int h) { texel = (1/w, 1/h); return new Size(w, h); }

        glProgram.use(); glProgram.setSamplerTexIdUniform("uTex", texId, 0);

      c = texture(uTex, v)
      // 1) probabilidad de piel por color, en YCbCr -rango clásico de la literatura, Chai y Ngan
